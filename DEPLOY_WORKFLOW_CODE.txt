name: Build and deploy YaarJam from flat ZIP

on:
  push:
    branches: [main, master]
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: github-pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 22

      - name: Verify flat release ZIP exists
        run: |
          set -eu
          test -f YaarJam_V1_1_ALL_FILES_NO_FOLDERS.zip
          unzip -t YaarJam_V1_1_ALL_FILES_NO_FOLDERS.zip

      - name: Extract flat files and assemble app folders
        run: |
          set -eu
          rm -rf _flat _app
          mkdir -p _flat _app/src/lib _app/src/components
          unzip -q YaarJam_V1_1_ALL_FILES_NO_FOLDERS.zip -d _flat

          cp _flat/package.json _app/package.json
          cp _flat/index.html _app/index.html
          cp _flat/tsconfig.json _app/tsconfig.json
          cp _flat/tsconfig.app.json _app/tsconfig.app.json
          cp _flat/tsconfig.node.json _app/tsconfig.node.json
          cp _flat/vite.config.ts _app/vite.config.ts

          cp _flat/App.tsx _app/src/App.tsx
          cp _flat/main.tsx _app/src/main.tsx
          cp _flat/styles.css _app/src/styles.css
          cp _flat/catalog.ts _app/src/catalog.ts
          cp _flat/catalog.test.ts _app/src/catalog.test.ts
          cp _flat/vite-env.d.ts _app/src/vite-env.d.ts

          cp _flat/supabase.ts _app/src/lib/supabase.ts
          cp _flat/rooms.ts _app/src/lib/rooms.ts
          cp _flat/tic-tac-toe.ts _app/src/lib/tic-tac-toe.ts
          cp _flat/tic-tac-toe-online.ts _app/src/lib/tic-tac-toe-online.ts
          cp _flat/tic-tac-toe.test.ts _app/src/lib/tic-tac-toe.test.ts
          cp _flat/tic-tac-toe-ai.ts _app/src/lib/tic-tac-toe-ai.ts
          cp _flat/tic-tac-toe-ai.test.ts _app/src/lib/tic-tac-toe-ai.test.ts

          cp _flat/ludo.ts _app/src/lib/ludo.ts
          cp _flat/ludo.test.ts _app/src/lib/ludo.test.ts
          cp _flat/ludo-online.ts _app/src/lib/ludo-online.ts
          cp _flat/ludo-online.test.ts _app/src/lib/ludo-online.test.ts
          cp _flat/party-prompts.ts _app/src/lib/party-prompts.ts
          cp _flat/party-prompts.test.ts _app/src/lib/party-prompts.test.ts
          cp _flat/truth-or-dare.ts _app/src/lib/truth-or-dare.ts
          cp _flat/truth-or-dare-online.ts _app/src/lib/truth-or-dare-online.ts
          cp _flat/truth-or-dare.test.ts _app/src/lib/truth-or-dare.test.ts
          cp _flat/truth-or-dare-online.test.ts _app/src/lib/truth-or-dare-online.test.ts

          cp _flat/RoomChat.tsx _app/src/components/RoomChat.tsx
          cp _flat/LocalPartyGamePanel.tsx _app/src/components/LocalPartyGamePanel.tsx
          cp _flat/LudoPanel.tsx _app/src/components/LudoPanel.tsx

      - name: Verify the four missing imports
        run: |
          set -eu
          test -f _app/src/lib/tic-tac-toe-ai.ts
          test -f _app/src/components/RoomChat.tsx
          test -f _app/src/components/LocalPartyGamePanel.tsx
          test -f _app/src/components/LudoPanel.tsx
          echo "All four missing imports exist."

      - name: Install dependencies
        working-directory: _app
        run: npm install --no-audit --no-fund

      - name: Run tests
        working-directory: _app
        run: npm test

      - name: Build production website
        working-directory: _app
        run: npm run build

      - uses: actions/configure-pages@v5
      - uses: actions/upload-pages-artifact@v3
        with:
          path: _app/dist

  deploy:
    needs: build
    runs-on: ubuntu-latest
    permissions:
      pages: write
      id-token: write
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - id: deployment
        uses: actions/deploy-pages@v4
